CREATE POLICY "Permitir cadastro de clientes" // Quem estiver sujeito ao RLS pode inserir uma linha em users, desde que ela tenha role = 'cliente'.
ON public.users
FOR INSERT
WITH CHECK (
    role = 'cliente'
);


CREATE POLICY "Usuário pode consultar seu próprio perfil"
ON public.users
FOR SELECT
USING (
    id = auth.uid()
);

ALTER TABLE public.users
ADD CONSTRAINT users_role_check
CHECK (
    role IN (
        'admin',
        'dono',
        'atendente',
        'tecnico',
        'funcionario'
    )
);

CREATE POLICY "Permitir que funcionários criem clientes" 
ON customer 
FOR INSERT 
TO authenticated 
WITH CHECK (
  -- Verifica se o ID de quem está logado (auth.uid()) possui a role correta na tabela users
  EXISTS (
    SELECT 1 
    FROM users 
    WHERE users.id_users = auth.uid() 
      AND users.role IN ('attendant', 'owner', 'admin')
  )
);
